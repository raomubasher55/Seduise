import { GoogleGenAI } from '@google/genai';

// Initialize the GoogleGenAI client.
// It automatically looks for the GEMINI_API_KEY environment variable.
const ai = new GoogleGenAI({
    apiKey: "AIzaSyDH1Rq7cMp_b8iMtqT4pwEXnHq3m7Y41N0"
});

/**
 * Sends a prompt to the Gemini model and prints the response.
 */
async function runGeminiExample() {
    try {
        const prompt = "hello";

        console.log(`Sending prompt: "${prompt}"`);

        // Call the model to generate content
        const response = await ai.models.generateContent({
            model: "gemini-2.5-flash",
            contents: [
                {
                    role: "user",
                    parts: [{ text: prompt }]
                }
            ],
            config:{
                systemInstruction: "your are coding agent",
            }
        });

        // The result is in response.text
        console.log("\n🤖 Gemini Response:");
        console.log(response.text);

    } catch (error) {
        console.error("An error occurred during the API call:", error);
    }
}

runGeminiExample();
